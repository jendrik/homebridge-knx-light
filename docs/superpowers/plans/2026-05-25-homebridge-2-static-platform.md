```ts
import type { PlatformConfig } from 'homebridge';

const DEFAULT_KNX_IP = '224.0.23.12';
const DEFAULT_KNX_PORT = 3671;

export interface LightDeviceConfig {
  readonly name: string;
  readonly setStatus: string;
  readonly listenStatus: string;
  readonly setBrightness?: string;
  readonly listenBrightness?: string;
  readonly setBrightnessR?: string;
  readonly listenBrightnessR?: string;
  readonly setBrightnessG?: string;
  readonly listenBrightnessG?: string;
  readonly setBrightnessB?: string;
  readonly listenBrightnessB?: string;
}

export interface LightMode {
  readonly dimmer: boolean;
  readonly rgb: boolean;
  readonly rgbw: boolean;
}

export interface InvalidDeviceConfig {
  readonly name: string;
  readonly reason: string;
}

export interface NormalizedPlatformConfig {
  readonly ip: string;
  readonly port: number;
  readonly devices: LightDeviceConfig[];
  readonly invalidDevices: InvalidDeviceConfig[];
}

type RawDeviceConfig = Record<string, unknown>;

export function normalizePlatformConfig(config: PlatformConfig | Record<string, unknown>): NormalizedPlatformConfig {
  const ip = readString(config.ip) ?? DEFAULT_KNX_IP;
  const port = readPort(config.port);
  const rawDevices = Array.isArray(config.devices) ? config.devices : [];
  const devices: LightDeviceConfig[] = [];
  const invalidDevices: InvalidDeviceConfig[] = [];

  for (const rawDevice of rawDevices) {
    const normalized = normalizeDeviceConfig(rawDevice as RawDeviceConfig);
    if ('reason' in normalized) {
      invalidDevices.push(normalized);
    } else {
      devices.push(normalized);
    }
  }

  return {
    ip,
    port,
    devices,
    invalidDevices,
  };
}

export function detectLightMode(device: LightDeviceConfig): LightMode {
  const hasRgb = Boolean(device.setBrightnessR && device.setBrightnessG && device.setBrightnessB);
  const hasRgbw = hasRgb && Boolean(device.setBrightness);

  return {
    dimmer: Boolean((device.setBrightness || device.listenBrightness) && !hasRgbw),
    rgb: hasRgb,
    rgbw: hasRgbw,
  };
}

function normalizeDeviceConfig(rawDevice: RawDeviceConfig): LightDeviceConfig | InvalidDeviceConfig {
  const name = readString(rawDevice.name) ?? 'Unnamed light';
  const setStatus = readString(rawDevice.set_status);
  const listenStatus = readString(rawDevice.listen_status);
  const missingFields = [
    ['set_status', setStatus],
    ['listen_status', listenStatus],
  ].filter(([, value]) => !value).map(([field]) => field);

  if (!readString(rawDevice.name)) {
    missingFields.unshift('name');
  }

  if (missingFields.length > 0 || !setStatus || !listenStatus) {
    return {
      name,
      reason: `missing required field(s): ${missingFields.join(', ')}`,
    };
  }

  return {
    name,
    setStatus,
    listenStatus,
    setBrightness: readString(rawDevice.set_brightness),
    listenBrightness: readString(rawDevice.listen_brightness),
    setBrightnessR: readString(rawDevice.set_brightness_r),
    listenBrightnessR: readString(rawDevice.listen_brightness_r),
    setBrightnessG: readString(rawDevice.set_brightness_g),
    listenBrightnessG: readString(rawDevice.listen_brightness_g),
    setBrightnessB: readString(rawDevice.set_brightness_b),
    listenBrightnessB: readString(rawDevice.listen_brightness_b),
  };
}

function readString(value: unknown): string | undefined {
  return typeof value === 'string' && value.trim().length > 0 ? value.trim() : undefined;
}

function readPort(value: unknown): number {
  if (typeof value === 'number' && Number.isInteger(value) && value > 0) {
    return value;
  }

  if (typeof value === 'string') {
    const parsed = Number(value);
    if (Number.isInteger(parsed) && parsed > 0) {
      return parsed;
    }
  }

  return DEFAULT_KNX_PORT;
}
```

```ts
import colorsys from 'colorsys';

export interface RgbValue {
  readonly r: number;
  readonly g: number;
  readonly b: number;
}

export interface RgbwValue extends RgbValue {
  readonly w: number;
}

export function hsvToRgb(hue: number, saturation: number, value: number): RgbValue {
  const rgb = colorsys.hsv2Rgb(hue, saturation, value);

  return {
    r: clampChannel(rgb.r),
    g: clampChannel(rgb.g),
    b: clampChannel(rgb.b),
  };
}

export function hsvToRgbw(hue: number, saturation: number, value: number): RgbwValue {
  const rgb = hsvToRgb(hue, saturation, value);
  const maxChannel = Math.max(rgb.r, rgb.g, rgb.b);

  if (maxChannel === 0) {
    return { r: 0, g: 0, b: 0, w: 0 };
  }

  const multiplier = 255 / maxChannel;
  const fullRed = rgb.r * multiplier;
  const fullGreen = rgb.g * multiplier;
  const fullBlue = rgb.b * multiplier;
  const maxFullChannel = Math.max(fullRed, fullGreen, fullBlue);
  const minFullChannel = Math.min(fullRed, fullGreen, fullBlue);
  const luminance = ((maxFullChannel + minFullChannel) / 2 - 127.5) * (255 / 127.5) / multiplier;

  return {
    r: clampChannel(rgb.r - luminance),
    g: clampChannel(rgb.g - luminance),
    b: clampChannel(rgb.b - luminance),
    w: clampChannel(luminance),
  };
}

function clampChannel(value: number): number {
  return Math.max(Math.min(Math.round(value), 255), 0);
}
```

```ts
import { API, StaticPlatformPlugin, Logger, PlatformConfig, AccessoryPlugin, Service, Characteristic, uuid } from 'homebridge';

import fakegato from 'fakegato-history';

import { Connection } from 'knx';

import { LightAccessory } from './accessory.js';
import { normalizePlatformConfig } from './config.js';

export class LightPlatform implements StaticPlatformPlugin {
  public readonly Service: typeof Service;
  public readonly Characteristic: typeof Characteristic;
  public readonly uuid: typeof uuid;

  public readonly fakeGatoHistoryService;
  public readonly connection: Connection;

  private readonly devices: LightAccessory[] = [];

  constructor(
    public readonly log: Logger,
    public readonly config: PlatformConfig,
    public readonly api: API,
  ) {
    this.Service = api.hap.Service;
    this.Characteristic = api.hap.Characteristic;
    this.uuid = api.hap.uuid;
    this.fakeGatoHistoryService = fakegato(this.api);

    const normalizedConfig = normalizePlatformConfig(config);

    this.connection = new Connection({
      ipAddr: normalizedConfig.ip,
      ipPort: normalizedConfig.port,
      handlers: {
        connected: () => {
          log.info('KNX connected');
        },
        error: (connstatus: unknown) => {
          log.error(`KNX status: ${String(connstatus)}`);
        },
      },
    });

    for (const invalidDevice of normalizedConfig.invalidDevices) {
      log.warn(`Skipping KNX light "${invalidDevice.name}": ${invalidDevice.reason}`);
    }

    for (const device of normalizedConfig.devices) {
      this.devices.push(new LightAccessory(this, device));
    }

    log.info(`Finished initializing ${this.devices.length} KNX light(s)`);
  }

  accessories(callback: (foundAccessories: AccessoryPlugin[]) => void): void {
    callback(this.devices);
  }
}
```

```ts
private writeColorState(rgbw: boolean): void {
  this.startColorFadeDebounce();

  if (this.hue === undefined || this.saturation === undefined || this.brightness === undefined) {

  }

    groupAddress: this.config.setBrightnessR,

  });
  const greenDatapoint = createDatapoint(this.platform.connection, {
    groupAddress: this.config.setBrightnessG,
    dpt: 'DPT5',

    groupAddress: this.config.setBrightnessB,
    dpt: 'DPT5',
  });

  if (rgbw) {
    if (!this.config.setBrightness) {
      this.platform.log.warn(`${this.config.name} cannot write RGBW color because white set address is missing`);
      return;
    }

    const rgbwValue = hsvToRgbw(this.hue, this.saturation, this.brightness);
    const whiteDatapoint = createDatapoint(this.platform.connection, {
      groupAddress: this.config.setBrightness,
      dpt: 'DPT5.001',
    });

    this.platform.log.info(`${this.config.name} set RGBW ${rgbwValue.r}/${rgbwValue.g}/${rgbwValue.b}/${rgbwValue.w}`);
    redDatapoint.write(rgbwValue.r);
    greenDatapoint.write(rgbwValue.g);
    blueDatapoint.write(rgbwValue.b);
    whiteDatapoint.write(rgbwValue.w);
    return;
  }

  const rgbValue = hsvToRgb(this.hue, this.saturation, this.brightness);
  this.platform.log.info(`${this.config.name} set RGB ${rgbValue.r}/${rgbValue.g}/${rgbValue.b}`);
  redDatapoint.write(rgbValue.r);
  greenDatapoint.write(rgbValue.g);
  blueDatapoint.write(rgbValue.b);
}

private startColorFadeDebounce(): void {
  this.timer = setTimeout(() => {
    this.timer = undefined;
  }, FADING_TIME_MS);
}
```

This implementation creates color set datapoints inside the shared write path to keep the refactor small. A later performance pass may cache those datapoints if real KNX traffic shows a measurable problem.
